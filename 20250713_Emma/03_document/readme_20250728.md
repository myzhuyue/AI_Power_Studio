## Inventory data

***Basic Formula***

```sql
期初库存 + 进货数量 - 期末库存 = 实际销量

或者

期初库存 + 进货数量 - 实际销量 = 期末库存
```

```sql
期初库存 = 
CALCULATE(
    SUM('库存表'[库存数量]),
    FILTER(
        ALL('库存表'[日期]),
        '库存表'[日期] = STARTOFMONTH('库存表'[日期])
    )
)
```

```sql
期末库存 = 
CALCULATE(
    SUM('库存表'[库存数量]),
    FILTER(
        ALL('库存表'[日期]),
        '库存表'[日期] = ENDOFMONTH('库存表'[日期])
    )
)
```

```sql
库存汇总表 = 
ADDCOLUMNS(
    SUMMARIZE(
        '库存表',
        '库存表'[商品],
        '库存表'[日期].[年份],
        '库存表'[日期].[月份],
        "库存月份", FORMAT('库存表'[日期], "yyyy年MM月")
    ),
    "期初库存", [期初库存],
    "期末库存", [期末库存]
)
```

## What if the start date and end date is not the actual and end date

当库存表中没有完整的每日数据时，我们需要先找到每个商品在每个月实际存在的最小日期（当月最早记录日期）和最大日期（当月最晚记录日期），再基于这些实际日期计算库存。可以使用以下 DAX 方案：

```sql
当月实际最小日期 = 
CALCULATE(
    MIN('库存表'[日期]),
    ALLEXCEPT('库存表', '库存表'[商品], '库存表'[日期].[年份], '库存表'[日期].[月份])
)
```
```
当月实际最大日期 = 
CALCULATE(
    MAX('库存表'[日期]),
    ALLEXCEPT('库存表', '库存表'[商品], '库存表'[日期].[年份], '库存表'[日期].[月份])
)
```

```
实际期初库存 = 
VAR CurrentMinDate = [当月实际最小日期]
RETURN
CALCULATE(
    SUM('库存表'[库存数量]),
    FILTER(
        ALLEXCEPT('库存表', '库存表'[商品], '库存表'[日期].[年份], '库存表'[日期].[月份]),
        '库存表'[日期] = CurrentMinDate
    )
)
```

```
实际期末库存 = 
VAR CurrentMaxDate = [当月实际最大日期]
RETURN
CALCULATE(
    SUM('库存表'[库存数量]),
    FILTER(
        ALLEXCEPT('库存表', '库存表'[商品], '库存表'[日期].[年份], '库存表'[日期].[月份]),
        '库存表'[日期] = CurrentMaxDate
    )
)
```

```
实际库存汇总表 = 
FILTER(
    ADDCOLUMNS(
        SUMMARIZE(
            '库存表',
            '库存表'[商品],
            '库存表'[日期].[年份],
            '库存表'[日期].[月份],
            "库存月份", FORMAT('库存表'[日期], "yyyy年MM月")
        ),
        "期初库存", [实际期初库存],
        "期末库存", [实际期末库存]
    ),
    NOT ISBLANK([期初库存]) && NOT ISBLANK([期末库存])
)
```
