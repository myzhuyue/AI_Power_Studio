load_time = 480

# 计算停机时间
df['downtime'] = df[['tpm', 'changeover', 'cleaning','meal_short_break', 'breakdown', 'process_issue', 
                     'manual_inspection', 'waiting_for_approval', 'waiting','missing_manpower','missing_materials', 
                     'project_downtime', 'others']].sum(axis=1)

# 计算时间稼动率
df['time_availability'] = (load_time - df['downtime']) / load_time

# 计算实际产量
df['actual_output'] = df['good_pieces'] + df['reject'] + df['rework']

# 计算理论加工周期
df['theoretical_cycle_time'] = 1 / df['ideal_run_rate']

# 计算稼动时间
df['operating_time'] = load_time - df['downtime']

# 计算性能稼动率
df['performance_rate'] = (df['actual_output'] * df['theoretical_cycle_time']) / df['operating_time']

# 计算良品率
df['quality_rate'] = df['good_pieces'] / df['actual_output']

# 计算 OEE
df['OEE'] = df['time_availability'] * df['performance_rate'] * df['quality_rate']
